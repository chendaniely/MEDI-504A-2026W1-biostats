(1) The column headers 0 to 28 are values of a day variable, not variable names.

tumor <- read_csv("data/tumorgrowth_wide.csv")

tumor_tidy <- tumor |>
  pivot_longer(
    `0`:last_col(),
    names_to = "day",
    names_transform = list(day = as.integer),
    values_to = "size",
    values_drop_na = TRUE
  )

Exercise 2: Cleaning a clinical trial dataset

opt |>
  count(Birth.outcome)
         Birth.outcome   n
1 Elective abortion      2

3 Lost to FU             9
4 Non-live birth        19

(1) The labels end in trailing spaces ("Live birth ").

opt |>
  filter(Birth.outcome == "Live birth") |>
  nrow()
[1] 0

(2) 0 rows, because "Live birth" doesn’t match "Live birth ".

opt_clean <- opt |>
  mutate(
    Birth.outcome = str_trim(Birth.outcome),
    Group = factor(Group, levels = c("C", "T"),
                   labels = c("control", "treatment"))
  ) |>
  filter(Birth.outcome == "Live birth") |>
  select(PID, Clinic, Group, Age, BMI, Birthweight, GA.at.outcome)

Exercise 3: Cytomegalovirus

cmv <- cytomegalovirus |>
  mutate(
    cmv = factor(cmv, levels = c(0, 1), labels = c("no", "yes")),
    recipient.cmv = factor(recipient.cmv, levels = c(0, 1),
                           labels = c("negative", "positive")),
    donor.cmv = factor(donor.cmv, levels = c(0, 1),
                       labels = c("negative", "positive"))
  )
