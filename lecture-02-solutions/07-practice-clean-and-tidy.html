opt_clean <- opt |>
  mutate(
    Birth.outcome = str_trim(Birth.outcome),
    Group = factor(Group, levels = c("C", "T"),
                   labels = c("control", "treatment"))

  filter(Birth.outcome == "Live birth") |>
  select(PID, Clinic, Group, Age, BMI, Birthweight, GA.at.outcome)
